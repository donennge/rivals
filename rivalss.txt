local function sfx(name)
    local list=SFX[name]; if not list then return end
    for _,s in ipairs(list) do
        task.delay(s.delay or 0, function()
            local snd=Instance.new("Sound")
            snd.SoundId=s.id; snd.Volume=s.vol or 0.5
            snd.PlaybackSpeed=s.speed or 1
            snd.Parent=SoundService; snd:Play()
            Debris:AddItem(snd,3)
        end)
    end
end

local function new(class, props, parent)
    local o=Instance.new(class)
    for k,v in pairs(props) do o[k]=v end
    o.Parent=parent; return o
end
local function tween(o,t,props,style,dir)
    local tw=TweenService:Create(o,TweenInfo.new(t,style or Enum.EasingStyle.Quint,dir or Enum.EasingDirection.Out),props)
    tw:Play(); return tw
end
local function corner(o,r) return new("UICorner",{CornerRadius=UDim.new(0,r)},o) end
local function stroke(o,color,th,tr)
    return new("UIStroke",{Color=color,Thickness=th or 1,Transparency=tr or 0,
        ApplyStrokeMode=Enum.ApplyStrokeMode.Border},o)
end
local function label(parent,props)
    local l=Instance.new("TextLabel")
    l.BackgroundTransparency=1; l.BorderSizePixel=0
    l.Font=Enum.Font.GothamMedium; l.TextColor3=C.text; l.TextSize=13
    for k,v in pairs(props) do l[k]=v end
    l.Parent=parent; return l
end
local function accentGradient(o)
    return new("UIGradient",{Color=ColorSequence.new({
        ColorSequenceKeypoint.new(0,C.accent),
        ColorSequenceKeypoint.new(0.5,C.purple),
        ColorSequenceKeypoint.new(1,C.blue),
    })},o)
end

local spinners={}
local function spin(g,speed) table.insert(spinners,{g=g,s=speed or 60}) end
RunService.RenderStepped:Connect(function(dt)
    for i=#spinners,1,-1 do
        local e=spinners[i]
        if e.g.Parent then e.g.Rotation=(e.g.Rotation+e.s*dt)%360
        else table.remove(spinners,i) end
    end
end)

local function shineText(lbl,period)
    period=period or 2.0
    lbl.TextColor3=WHITE
    local g=new("UIGradient",{Color=ColorSequence.new({
        ColorSequenceKeypoint.new(0,C.red),
        ColorSequenceKeypoint.new(0.4,C.red),
        ColorSequenceKeypoint.new(0.5,WHITE),
        ColorSequenceKeypoint.new(0.6,C.red),
        ColorSequenceKeypoint.new(1,C.red),
    }),Offset=Vector2.new(1,0)},lbl)
    task.spawn(function()
        while g.Parent do
            g.Offset=Vector2.new(1,0)
            tween(g,period*0.65,{Offset=Vector2.new(-1,0)},Enum.EasingStyle.Sine,Enum.EasingDirection.InOut)
            task.wait(period)
        end
    end)
    return g
end

local function makeToggle(parent,ctrl,order)
    local row=new("TextButton",{
        Size=UDim2.new(1,0,0,34),
        BackgroundColor3=C.hover, BackgroundTransparency=1,
        AutoButtonColor=false, Text="", ClipsDescendants=true,
        LayoutOrder=order,
    },parent)
    corner(row,8)
    local hbar=new("Frame",{
        AnchorPoint=Vector2.new(0,0.5), Position=UDim2.new(0,0,0.5,0),
        Size=UDim2.new(0,3,0,0), BackgroundColor3=C.accent, BorderSizePixel=0,
    },row)
    local name=label(row,{
        Text=ctrl.title, Position=UDim2.new(0,10,0,0),
        Size=UDim2.new(1,-60,1,0),
        TextXAlignment=Enum.TextXAlignment.Left,
    })
    local sw=new("Frame",{
        AnchorPoint=Vector2.new(1,0.5), Position=UDim2.new(1,-8,0.5,0),
        Size=UDim2.new(0,38,0,20), BackgroundColor3=C.line, BorderSizePixel=0,
    },row)
    corner(sw,10)
    local swStroke=stroke(sw,C.accent,2,1)
    local knob=new("Frame",{
        AnchorPoint=Vector2.new(0,0.5), Position=UDim2.new(0,3,0.5,0),
        Size=UDim2.new(0,14,0,14), BackgroundColor3=C.text, BorderSizePixel=0,
    },sw)
    corner(knob,7)
    local function set(v,silent)
        tween(sw,0.25,{BackgroundColor3=v and C.accent or C.line})
        tween(swStroke,0.25,{Transparency=v and 0.45 or 1})
        tween(knob,0.32,{Position=UDim2.new(0,v and 21 or 3,0.5,0)},Enum.EasingStyle.Back)
        tween(name,0.2,{TextColor3=v and C.text or C.muted})
        fire(ctrl.id,v,silent)
        if not silent then sfx(v and "on" or "off") end
    end
    row.MouseEnter:Connect(function()
        tween(row,0.15,{BackgroundTransparency=0.45})
        tween(hbar,0.2,{Size=UDim2.new(0,3,0,16)})
        sfx("hover")
    end)
    row.MouseLeave:Connect(function()
        tween(row,0.15,{BackgroundTransparency=1})
        tween(hbar,0.2,{Size=UDim2.new(0,3,0,0)})
    end)
    row.Activated:Connect(function(input)
        local relX=0.85
        if input and input.Position then
            relX=math.clamp((input.Position.X-row.AbsolutePosition.X)/row.AbsoluteSize.X,0,1)
        end
        ripple(row,relX)
        set(not Flags[ctrl.id])
    end)
    set(ctrl.default,true)
    name.TextColor3=ctrl.default and C.text or C.muted
end

local function makeSlider(parent,ctrl,order)
    local row=new("Frame",{Size=UDim2.new(1,0,0,48),BackgroundTransparency=1,LayoutOrder=order},parent)
    label(row,{Text=ctrl.title,Position=UDim2.new(0,10,0,4),Size=UDim2.new(1,-70,0,18),
        TextXAlignment=Enum.TextXAlignment.Left})
    local valueLabel=label(row,{
        Text="", Font=Enum.Font.GothamBold, TextColor3=C.accent,
        AnchorPoint=Vector2.new(1,0), Position=UDim2.new(1,-8,0,4),
        Size=UDim2.new(0,60,0,18), TextXAlignment=Enum.TextXAlignment.Right,
    })
    local hit=new("TextButton",{
        Position=UDim2.new(0,10,0,26), Size=UDim2.new(1,-20,0,18),
        BackgroundTransparency=1, Text="", AutoButtonColor=false,
    },row)
    local bar=new("Frame",{
        AnchorPoint=Vector2.new(0,0.5), Position=UDim2.new(0,0,0.5,0),
        Size=UDim2.new(1,0,0,6), BackgroundColor3=C.line, BorderSizePixel=0,
    },hit)
    corner(bar,3)
    local fill=new("Frame",{Size=UDim2.new(0,0,1,0),BackgroundColor3=WHITE,BorderSizePixel=0},bar)
    corner(fill,3); accentGradient(fill)
    local knob=new("Frame",{
        AnchorPoint=Vector2.new(0.5,0.5), Size=UDim2.new(0,14,0,14),
        BackgroundColor3=C.text, BorderSizePixel=0,
    },bar)
    corner(knob,7)
    local knobStroke=stroke(knob,C.accent,2,1)
    local min,max=ctrl.min,ctrl.max
    local mult=10^ctrl.decimals
    local function setValue(v,silent)
        v=math.floor(math.clamp(v,min,max)*mult+0.5)/mult
        local a=(v-min)/(max-min)
        tween(fill,0.08,{Size=UDim2.new(a,0,1,0)},Enum.EasingStyle.Linear)
        tween(knob,0.08,{Position=UDim2.new(a,0,0.5,0)},Enum.EasingStyle.Linear)
        valueLabel.Text=tostring(v)
        fire(ctrl.id,v,silent)
    end
    local function fromX(x)
        local a=math.clamp((x-bar.AbsolutePosition.X)/bar.AbsoluteSize.X,0,1)
        setValue(min+(max-min)*a)
    end
    local dragging=false
    hit.InputBegan:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then
            dragging=true
            sfx("select")
            tween(knob,0.18,{Size=UDim2.new(0,19,0,19)},Enum.EasingStyle.Back)
            tween(knobStroke,0.18,{Transparency=0.3})
            fromX(i.Position.X)
            i.Changed:Connect(function()
                if i.UserInputState==Enum.UserInputState.End then
                    dragging=false
                    tween(knob,0.18,{Size=UDim2.new(0,14,0,14)})
                    tween(knobStroke,0.18,{Transparency=1})
                end
            end)
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType==Enum.UserInputType.MouseMovement or i.UserInputType==Enum.UserInputType.Touch) then
            fromX(i.Position.X)
        end
    end)
    setValue(ctrl.default,true)
end

local function makeDraggable(handle,target,opts)
    opts=opts or {}
    local drag={active=false}
    local goal,cur=Vector2.zero,Vector2.zero
    local startPos,startAbs
    handle.InputBegan:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then
            drag.active=true; drag.input=i; drag.start=i.Position; drag.moved=0
            startPos=target.Position; startAbs=target.AbsolutePosition
            goal,cur=Vector2.zero,Vector2.zero
            i.Changed:Connect(function()
                if i.UserInputState==Enum.UserInputState.End then
                    drag.active=false
                    if opts.onTap and drag.moved<8 then opts.onTap() end
                end
            end)
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if not drag.active then return end
        local isMine=i==drag.input
            or (i.UserInputType==Enum.UserInputType.MouseMovement and drag.input.UserInputType==Enum.UserInputType.MouseButton1)
        if not isMine then return end
        local d=Vector2.new(i.Position.X-drag.start.X,i.Position.Y-drag.start.Y)
        drag.moved=math.max(drag.moved,d.Magnitude)
        local vp,sz=gui.AbsoluteSize,target.AbsoluteSize
        local minX,maxX,minY,maxY
        if opts.full then
            minX,maxX=6-startAbs.X,vp.X-sz.X-6-startAbs.X
            minY,maxY=6-startAbs.Y,vp.Y-sz.Y-6-startAbs.Y
        else
            minX,maxX=140-startAbs.X-sz.X,vp.X-140-startAbs.X
            minY,maxY=-startAbs.Y,vp.Y-56-startAbs.Y
        end
        goal=Vector2.new(math.clamp(d.X,minX,maxX),math.clamp(d.Y,minY,maxY))
    end)
    RunService.RenderStepped:Connect(function(dt)
        if not startPos or (not drag.active and (cur-goal).Magnitude<0.05) then return end
        cur=cur:Lerp(goal,1-math.exp(-dt*26))
        target.Position=UDim2.new(startPos.X.Scale,startPos.X.Offset+cur.X,
            startPos.Y.Scale,startPos.Y.Offset+cur.Y)
    end)
end

-- Fly / Noclip / Speed
local function Fly()
    local bv, currentHRP
    local function cleanup()
        if bv then bv:Destroy() bv = nil end
        currentHRP = nil
    end
    while flying do
        local char = LP.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then cleanup() task.wait(0.1) else
            if hrp ~= currentHRP then
                cleanup() currentHRP = hrp
                bv = Instance.new("BodyVelocity")
                bv.MaxForce = Vector3.new(1e5, 1e5, 1e5)
                bv.Parent = hrp
            end
            local move = Vector3.zero
            if UIS:IsKeyDown(Enum.KeyCode.W) then move += Camera.CFrame.LookVector end
            if UIS:IsKeyDown(Enum.KeyCode.S) then move -= Camera.CFrame.LookVector end
            if UIS:IsKeyDown(Enum.KeyCode.A) then move -= Camera.CFrame.RightVector end
            if UIS:IsKeyDown(Enum.KeyCode.D) then move += Camera.CFrame.RightVector end
            if UIS:IsKeyDown(Enum.KeyCode.Space) then move += Vector3.new(0, 1, 0) end
            if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then move -= Vector3.new(0, 1, 0) end
            bv.Velocity = move * 60
            task.wait()
        end
    end
    cleanup()
end

local function isEnemy(plr)
    if not plr or plr == LP then return false end
    local myT = LP:GetAttribute("TeamID")
    local thrT = plr:GetAttribute("TeamID")
    if myT ~= nil and thrT ~= nil then return tostring(myT) ~= tostring(thrT) end
    if LP.Team and plr.Team then return LP.Team ~= plr.Team end
    return true
end

local function getValidHead(char)
    local head = char:FindFirstChild("Head")
    if not head then return nil end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {LP.Character, Camera}
    local result = Workspace:Raycast(Camera.CFrame.Position, (head.Position - Camera.CFrame.Position).Unit * 1000, params)
    return (result == nil or result.Instance:IsDescendantOf(char)) and head or nil
end

        fovCircle.Size = UDim2.new(0, calculatedRadius * 2, 0, calculatedRadius * 2)
        stroke.Color = rgb
    end
    local bestHead = nil
    local shortestDist = 9e9
    for _, plr in pairs(Players:GetPlayers()) do
        local tracer = tracers[plr]
        local char = plr.Character
        if char and isEnemy(plr) then
            local head = char:FindFirstChild("Head")
            local torso = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso") or char:FindFirstChild("HumanoidRootPart")
            local highlight = char:FindFirstChild("Highlight") or Instance.new("Highlight", char)
            if head and torso then
                local distFromMe = (head.Position - Camera.CFrame.Position).Magnitude
                local headPos, headOnScreen = Camera:WorldToViewportPoint(head.Position)
                local torsoPos, torsoOnScreen = Camera:WorldToViewportPoint(torso.Position)
                local fovDist = (Vector2.new(headPos.X, headPos.Y) - screenCenter).Magnitude
                if tracer then
                    tracer.Visible = (tracersEnabled and torsoOnScreen and distFromMe <= maxDistance)
                    if tracer.Visible then
                        tracer.From = screenCenter
                        tracer.To = Vector2.new(torsoPos.X, torsoPos.Y)
                        tracer.Color = rgb
                    end
                end
                highlight.Enabled = chamsEnabled
                highlight.FillColor = rgb
                if headOnScreen and distFromMe <= maxDistance then
                    if softLockEnabled then
                        if fovDist <= calculatedRadius and fovDist < shortestDist then
                            local valid = getValidHead(char)
                            if valid then shortestDist = fovDist; bestHead = valid end
                        end
                    elseif silentAimEnabled then
                        if fovDist < shortestDist then
                            local valid = getValidHead(char)
                            if valid then shortestDist = fovDist; bestHead = valid end
                        end
                    end
                end
            end
        elseif tracer then
            tracer.Visible = false
        end
    end
    activeTargetPos = bestHead and bestHead.Position or nil
end)

local old
old = hookmetamethod(game, "__index", newcclosure(function(self, idx)
    if (silentAimEnabled or softLockEnabled) and activeTargetPos and not checkcaller() and idx == "ViewportSize" and self == Camera then
        local pos, on = Camera:WorldToViewportPoint(activeTargetPos)
        if on then return Vector2.new(pos.X * 2, pos.Y * 2) end
    end
    return old(self, idx)
end))

-- Unlocker Function
local function runUnlocker()
    notify("Executing Unlocker...", C.accent)
    task.spawn(function()
        local _stbl;
        _stbl = hookfunction(getrenv().setmetatable, newcclosure(function(tbl, mt)
            if mt and typeof(mt) == "table" and rawget(mt, "__mode") == "kv" then
                local tr = debug.traceback()
                if tr:find("MiscellaneousController") then return _stbl({1,2,3}, {}) end
            end
            return _stbl(tbl, mt)
        end))
        
        pcall(function()
            local _plrs = game:GetService("Players")
            local _rs = game:GetService("ReplicatedStorage")
            local _http = game:GetService("HttpService")
            local _run = game:GetService("RunService")
            local _lp = _plrs.LocalPlayer
            local _pscripts = _lp.PlayerScripts
            local _ctrl = _pscripts.Controllers
            local _mods = _rs:WaitForChild("Modules", 10)
            local _enumLib = require(_mods:WaitForChild("EnumLibrary", 10))
            if _enumLib then pcall(function() _enumLib:WaitForEnumBuilder() end) end
            local _cosLib = require(_mods:WaitForChild("CosmeticLibrary", 10))
            local _datCtrl = require(_ctrl:WaitForChild("PlayerDataController", 10))
            local _eq, _favs = {}, {}
            local _buildingWep = nil
            local _fakeInv = {}
            
            local function _mkCosmetic(nm, ctype, opts)
                local _base = _cosLib.Cosmetics[nm]
                if not _base then return nil end
                local _d = {}
                for k, v in pairs(_base) do _d[k] = v end
                _d.Name = nm; _d.Type = _d.Type or ctype; _d.Seed = _d.Seed or math.random(1, 1000000)
                if _enumLib then
                    local _s, _eid = pcall(_enumLib.ToEnum, _enumLib, nm)
                    if _s and _eid then _d.Enum = _eid; _d.ObjectID = _d.ObjectID or _eid end
                end
                if opts then
                    if opts.inverted ~= nil then _d.Inverted = opts.inverted end
                    if opts.favoritesOnly ~= nil then _d.OnlyUseFavorites = opts.favoritesOnly end
                end
                return _d
            end
            
            local _cosTypes = {"Skin","Wrap","Charm","Dance","Emote"}
            local function _isCosType(cosObj)
                if not cosObj then return false end
                for _, t in ipairs(_cosTypes) do if cosObj.Type == t then return true end end
                return false
            end
            
            _cosLib.OwnsCosmeticNormally = function() return true end
            _cosLib.OwnsCosmeticUniversally = function() return true end
            _cosLib.OwnsCosmeticForWeapon = function() return true end
            local _origOwns = _cosLib.OwnsCosmetic
            _cosLib.OwnsCosmetic = function(self, inv, nm, wep)
                if nm:find("MISSING_") or nm == "Bubble Gun" then return _origOwns(self, inv, nm, wep) end
                local c = _cosLib.Cosmetics[nm]
                if c and _isCosType(c) then return true end
                return _origOwns(self, inv, nm, wep)
            end
            
            local _origGet = _datCtrl.Get
            _datCtrl.Get = function(self, key)
                local _val = _origGet(self, key)
                if key == "CosmeticInventory" then
                    local _prx = {}
                    if _val then for k, v in pairs(_val) do if _isCosType(_cosLib.Cosmetics[k]) then _prx[k] = v end end end
                    return setmetatable(_prx, { __index = function(t, k) if _isCosType(_cosLib.Cosmetics[k]) then return true end return nil end })
                end
                return _val
            end
            
            local _origGetWep = _datCtrl.GetWeaponData
            _datCtrl.GetWeaponData = function(self, wn)
                local _d = _origGetWep(self, wn)
                if not _d then return nil end
                local _m = {}
                for k, v in pairs(_d) do _m[k] = v end
                _m.Name = wn
                if _eq[wn] then for ct, cd in pairs(_eq[wn]) do _m[ct] = cd end end
                return _m
            end
            
            if hookmetamethod then
                local _remotes = _rs:FindFirstChild("Remotes")
                local _dataRem = _remotes and _remotes:FindFirstChild("Data")
                local _equipRem = _dataRem and _dataRem:FindFirstChild("EquipCosmetic")
                if _equipRem then
                    local _onc
                    _onc = hookmetamethod(game, "__namecall", function(self, ...)
                        if getnamecallmethod() ~= "FireServer" then return _onc(self, ...) end
                        local _a = {...}
                        if self == _equipRem then
                            local _wn = _a[1]; local _ct = _a[2]; local _cn = _a[3]; local _opts = _a[4] or {}
                            _eq[_wn] = _eq[_wn] or {}
                            if not _cn or _cn == "None" or _cn == "" then
                                _eq[_wn][_ct] = nil
                                if not next(_eq[_wn]) then _eq[_wn] = nil end
                            else
                                local _cloned = _mkCosmetic(_cn, _ct, { inverted = _opts.IsInverted, favoritesOnly = _opts.OnlyUseFavorites })
                                if _cloned then _eq[_wn][_ct] = _cloned end
                            end
                            task.defer(function() pcall(function() _datCtrl.CurrentData:Replicate("WeaponInventory") end) end)
                            return
                        end
                        return _onc(self, ...)
                    end)
                end
            end
            
            local _cliItem
            pcall(function() _cliItem = require(_lp.PlayerScripts.Modules.ClientReplicatedClasses.ClientFighter.ClientItem) end)
            if _cliItem and _cliItem._CreateViewModel then
                local _origCVM = _cliItem._CreateViewModel
                _cliItem._CreateViewModel = function(self, vmRef)
                    local _wn = self.Name
                    local _wp = self.ClientFighter and self.ClientFighter.Player
                    _buildingWep = (_wp == _lp) and _wn or nil
                    if _wp == _lp and _eq[_wn] then
                        local _dk = self:ToEnum("Data")
                        if vmRef[_dk] then
                            if _eq[_wn].Skin then vmRef[_dk][self:ToEnum("Skin")] = _eq[_wn].Skin; vmRef[_dk][self:ToEnum("Name")] = _eq[_wn].Skin.Name end
                            if _eq[_wn].Charm then vmRef[_dk][self:ToEnum("Charm")] = _eq[_wn].Charm end
                            if _eq[_wn].Wrap then vmRef[_dk][self:ToEnum("Wrap")] = _eq[_wn].Wrap end
                        elseif vmRef.Data then
                            if _eq[_wn].Skin then vmRef.Data.Skin = _eq[_wn].Skin; vmRef.Data.Name = _eq[_wn].Skin.Name end
                            if _eq[_wn].Charm then vmRef.Data.Charm = _eq[_wn].Charm end
                            if _eq[_wn].Wrap then vmRef.Data.Wrap = _eq[_wn].Wrap end
                        end
                    end
                    local _r = _origCVM(self, vmRef)
                    _buildingWep = nil
                    return _r
                end
            end
        end)
        notify("All skins/wraps/charms unlocked!", C.green)
    end)
end
